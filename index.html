#!/usr/bin/env python3 """ Batch resize images to 300x300 square, cropping from center. Usage: python resize_headshots.py /path/to/image/folder """ import sys import os from PIL import Image def resize_to_square(image_path, output_path, size=300): """Crop to center square and resize to specified size.""" with Image.open(image_path) as img: # Convert to RGB if needed (handles PNG with transparency) if img.mode != 'RGB': img = img.convert('RGB') # Get dimensions width, height = img.size # Crop to center square if width > height: # Wider than tall - crop sides left = (width - height) // 2 top = 0 right = left + height bottom = height else: # Taller than wide - crop top/bottom left = 0 top = (height - width) // 2 right = width bottom = top + width img_cropped = img.crop((left, top, right, bottom)) # Resize to target size img_resized = img_cropped.resize((size, size), Image.Resampling.LANCZOS) # Save img_resized.save(output_path, 'JPEG', quality=90) print(f"✓ Processed: {os.path.basename(image_path)}") def main(): if len(sys.argv) < 2: print("Usage: python resize_headshots.py /path/to/images") sys.exit(1) folder = sys.argv[1] if not os.path.isdir(folder): print(f"Error: {folder} is not a directory") sys.exit(1) # Create output folder output_folder = os.path.join(folder, "resized_300x300") os.makedirs(output_folder, exist_ok=True) # Process all jpg/jpeg files for filename in os.listdir(folder): if filename.lower().endswith(('.jpg', '.jpeg')): input_path = os.path.join(folder, filename) output_path = os.path.join(output_folder, filename) try: resize_to_square(input_path, output_path, size=300) except Exception as e: print(f"✗ Error processing {filename}: {e}") print(f"\nDone! Resized images saved to: {output_folder}") if __name__ == "__main__": main()
